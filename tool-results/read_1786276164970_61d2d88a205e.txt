   254→   254→async function findMobileLoginButton() {
   255→   255→  // Strategy 1: find clickable element by text content
   256→   256→  const result = await page.evaluate((texts) => {
   257→   257→    const lowerTexts = texts.map((t) => t.toLowerCase());
   258→   258→    const clickableSelectors = 'button, a, div[role="button"], span[role="button"], [class*="btn" i], [class*="button" i]';
   259→   259→
   260→   260→    const candidates = Array.from(document.querySelectorAll(clickableSelectors));
   261→   261→    for (const el of candidates) {
   262→   262→      // Get text from element AND its children, trimmed
   263→   263→      const text = (el.innerText || el.textContent || '').trim().toLowerCase();
   264→   264→      if (!text || text.length > 100) continue; // skip huge elements / empty
   265→   265→
   266→   266→      const matches = lowerTexts.some((t) => text === t || text.includes(t));
   267→   267→      if (!matches) continue;
   268→   268→
   269→   269→      // Skip elements with very small size (icons, hidden)
   270→   270→      const rect = el.getBoundingClientRect();
   271→   271→      if (rect.width < 100 || rect.height < 20) continue;
   272→   272→      if (rect.top < 0 || rect.left < 0) continue;
   273→   273→
   274→   274→      // Skip if text contains "continue with google" or other non-mobile
   275→   275→      if (text.includes('google') || text.includes('email') || text.includes('tiktok') || text.includes('facebook')) {
   276→   276→        continue;
   277→   277→      }
   278→   278→
   279→   279→      // Found it! Return selector info
   280→   280→      el.setAttribute('data-capcut-mobile-btn', 'true');
   281→   281→      return {
   282→   282→        found: true,
   283→   283→        text: text.slice(0, 80),
   284→   284→        tag: el.tagName.toLowerCase(),
   285→   285→        width: rect.width,
   286→   286→        height: rect.height,
   287→   287→        top: rect.top,
   288→   288→        left: rect.left,
   289→   289→      };
   290→   290→    }
   291→   291→    return { found: false };
   292→   292→  }, MOBILE_BTN_TEXTS);
   293→   293→
   294→   294→  return result;
   295→   295→}

   297→   297→async function findQrCanvasOrImg() {

   301→   301→    const canvases = Array.from(document.querySelectorAll('canvas'));
   302→   302→    for (const c of canvases) {
   303→   303→      const rect = c.getBoundingClientRect();
   304→   304→      // QR codes are usually square, 100-400px
   305→   305→      if (Math.abs(rect.width - rect.height) < 20 && rect.width >= 100 && rect.width <= 500) {
   306→   306→        // Skip if it's the CapCut logo or something
   307→   307→        const parent = c.parentElement;
   308→   308→        const parentClass = parent ? parent.className.toLowerCase() : '';
   309→   309→        if (parentClass.includes('logo') || parentClass.includes('icon')) continue;

   312→   312→          type: 'canvas',
   313→   313→          width: rect.width,
   314→   314→          height: rect.height,
   315→   315→          top: rect.top,
   316→   316→          left: rect.left,
   317→   317→          parentClass: parentClass.slice(0, 80),

   420→   420→        }
   421→   421→      }
   422→   422→
   423→   423→      // Remove the data attribute for cleanliness
   424→   424→      await page.evaluate(() => {
   425→   425→        const btn = document.querySelector('[data-capcut-mobile-btn="true"]');
   426→   426→        if (btn) btn.removeAttribute('data-capcut-mobile-btn');
   427→   427→      });
   428→   428→
   429→   429→      if (mobileBtnClicked) break;
   430→   430→    }
   431→   431→  } catch (e) {
   432→   432→    logger.warn({ err: e.message, attempt }, 'Error finding mobile button');
   433→   433→  }
   434→   434→  if (!mobileBtnClicked) {
   435→   435→    if (attempt === 0) logger.warn('CapCut Mobile button not found yet, retrying...');
   436→   436→    await new Promise((r) => setTimeout(r, 2000));
   437→   437→  }
   438→   438→}
   439→   439→
   440→   440→if (!mobileBtnClicked) {
   441→   441→  logger.warn('Could not find "Continue with CapCut Mobile" button after 8 attempts.');
   442→   442→  logger.info('Continuing anyway — will try to detect QR if it auto-appears...');
   443→   443→} else {
   444→   444→  logger.info('Waiting for QR code modal to render...');
   445→   445→  await new Promise((r) => setTimeout(r, 3000));
   446→   446→
   447→   447→  // Diagnostic: check what changed after click
   448→   448→  const diag = await page.evaluate(() => {
   449→   449→    const allPages = window.length; // number of frames
   450→   450→    const canvases = document.querySelectorAll('canvas').length;
   451→   451→    const iframes = document.querySelectorAll('iframe').length;
   452→   452→    const dialogs = document.querySelectorAll('[role="dialog"], [class*="modal" i], [class*="popup" i], [class*="dialog" i]').length;
   453→   453→    // Check for any newly-visible overlay

   455→   455→      .filter((d) => {

   457→   457→        const rect = d.getBoundingClientRect();
   458→   458→        return (
   459→   459→          (s.position === 'fixed' || s.position === 'absolute') &&
   460→   460→          rect.width > 200 &&
   461→   461→          rect.height > 200 &&
   462→   462→          s.zIndex !== 'auto' &&
   463→   463→          parseInt(s.zIndex || '0', 10) > 100
   464→   464→        );
   465→   465→      }).length;
   466→   466→    return { allPages, canvases, iframes, dialogs, overlays, url: location.href };
   467→   467→  }).catch(() => ({}));
   468→   468→  logger.info({ diag }, 'Post-click diagnostics (canvases, iframes, dialogs, overlays)');
   469→   469→
   470→   470→  // List all iframes with their src + position (CapCut Mobile login uses an iframe!)
   471→   471→  const iframeInfos = await page.evaluate(() => {

   473→   473→      const r = f.getBoundingClientRect();

   526→   526→
   527→   527→// ====== Polling: detect login + capture QR ======
   528→   528→const startTime = Date.now();
   529→   529→let lastScreenshotTime = 0;
   530→   530→let qrFound = false;
   531→   531→let tries = 0;
   532→   532→
   533→   533→async function checkLogin() {
   534→   534→  tries++;
   535→   535→  const elapsedMs = Date.now() - startTime;
   536→   536→
   537→   537→  if (elapsedMs > MAX_WAIT_MS) {
   538→   538→    logger.warn({ elapsedMs }, `Timeout ${MAX_WAIT_MS / 1000}s. No login detected.`);
   539→   539→    return 'timeout';
   540→   540→  }
   541→   541→
   542→   542→  try {
   543→   543→    // Use popupPage if it exists (where QR code lives), otherwise main page
   544→   544→    const activePage = popupPage && !popupPage.isClosed() ? popupPage : page;
   545→   545→    const currentUrl = activePage.url();
   546→   546→
   547→   547→    // Cek 1: URL berubah dari /login atau /scan-qr-code (redirect ke home/dashboard)
   548→   548→    const stillOnLogin = /\/login|\/scan-qr-code/i.test(currentUrl);
   549→   549→
   550→   550→    // Cek 2: cookies session (check both pages, cookies are shared per profile)
   551→   551→    const cookies = await page.cookies();
   552→   552→    const sessionCookie = cookies.find(
   553→   553→      (c) =>
   554→   554→        /session|token|uid|passport|login|sid|tt_csrf|s_v_web_id/i.test(c.name) && c.value && c.value.length > 10
   555→   555→    );
   556→   556→
   557→   557→    // Cek 3: avatar / user element di header (check main page)
   558→   558→    let avatarFound = false;
   559→   559→    if (!stillOnLogin) {
   560→   560→      try {
   561→   561→        avatarFound = await page.$(
   562→   562→          '[class*="avatar" i], [data-e2e*="user" i], [data-testid*="user" i], [class*="user-info" i], [class*="header-user" i], [class*="account" i]'
   563→   563→        ) ? true : false;
   564→   564→      } catch (_) {}
   565→   565→    }
   566→   566→
   567→   567→    if (sessionCookie && (!stillOnLogin || avatarFound)) {
   568→   568→      logger.info({ url: currentUrl, cookie: sessionCookie?.name, avatarFound }, 'LOGIN SUCCESS detected!');
   569→   569→      return 'success';
   570→   570→    }
   571→   571→
   572→   572→    // Screenshot QR code — prefer popupPage (where QR actually is)
   573→   573→    const now = Date.now();
   574→   574→    if (now - lastScreenshotTime > 1500) {
   575→   575→      try {
   576→   576→        if (popupPage && !popupPage.isClosed()) {
   577→   577→          // Screenshot the popup page (QR code lives here!)
   578→   578→          await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   579→   579→          if (!qrFound) {
   580→   580→            logger.info({ popupUrl: popupPage.url() }, 'Capturing QR from popup page');
   581→   581→            qrFound = true;
   582→   582→          }
   583→   583→        } else {
   584→   584→          // Fallback: try to find QR element on main page
   585→   585→          const found = await findQrCanvasOrImg();
   586→   586→          if (found && found.found) {
   587→   587→            if (!qrFound) {
   588→   588→              logger.info(
   589→   589→                { type: found.type, w: Math.round(found.width), h: Math.round(found.height) },
   590→   590→                'QR code element detected on main page, capturing...'
   591→   591→              );
   592→   592→              qrFound = true;
   593→   593→            }
   594→   594→            await page.screenshot({
   595→   595→              path: qrLatestPath,
   596→   596→              clip: {
   597→   597→                x: Math.max(0, found.left - 30),
   598→   598→                y: Math.max(0, found.top - 30),
   599→   599→                width: found.width + 60,
   600→   600→                height: found.height + 60,
   601→   601→              },
   602→   602→            });
   603→   603→          } else {
   604→   604→            await page.screenshot({ path: qrLatestPath, fullPage: false });
   605→   605→          }
   606→   606→        }
   607→   607→        lastScreenshotTime = now;
   608→   608→
   609→   609→        // Save periodic debug screenshot
   610→   610→        if (tries % 5 === 0) {
   611→   611→          const debugPath = path.join(screenshotDir, `debug-${tries}.png`);
   612→   612→          await activePage.screenshot({ path: debugPath, fullPage: false }).catch(() => {});
   613→   613→        }
   614→   614→      } catch (e) {
   615→   615→        logger.warn({ err: e.message }, 'Screenshot error');
   616→   616→      }
   617→   617→    }
   618→   618→
   619→   619→    if (tries % 6 === 0) {
   620→   620→      logger.info(
   621→   621→        { tries, elapsedS: Math.round(elapsedMs / 1000), url: currentUrl, qrFound, hasSessionCookie: !!sessionCookie, mobileBtnClicked, popupAlive: popupPage && !popupPage.isClosed() },
   622→   622→        'Waiting for login...'
   623→   623→      );
   624→   624→    }
   625→   625→  } catch (e) {
   626→   626→    logger.warn({ err: e.message }, 'Polling error');
   627→   627→  }
   628→   628→
   629→   629→  return 'continue';
   630→   630→}
   631→   631→
   632→   632→const pollLoop = async () => {
   633→   633→  while (true) {
   634→   634→    const result = await checkLogin();
   635→   635→    if (result === 'success') {
   636→   636→      // Beri delay sedikit untuk pastikan session tersimpan
   637→   637→      await new Promise((r) => setTimeout(r, 2000));
   638→   638→      try {
   639→   639→        await browser.close();
   640→   640→      } catch (_) {}
   641→   641→      httpServer.close();
   642→   642→      logger.info(`✓ Session saved to ${userDataDir}`);
   643→   643→      logger.info('Set CAPCUT_USER_DATA_DIR=./.capcut-profile di .env untuk reuse session ini.');
   644→   644→      process.exit(0);
   645→   645→    }
   646→   646→    if (result === 'timeout') {
   647→   647→      try {
   648→   648→        await browser.close();
   649→   649→      } catch (_) {}
   650→   650→      httpServer.close();
   651→   651→      process.exit(1);
   652→   652→    }
   653→   653→    await new Promise((r) => setTimeout(r, POLL_INTERVAL_MS));
   654→   654→  }
   655→   655→};
   656→   656→
   657→   657→pollLoop().catch((e) => {
   658→   658→  logger.error({ err: e.message }, 'Fatal error in poll loop');
   659→   659→  try { browser.close(); } catch (_) {}
   660→   660→  httpServer.close();
   661→   661→  process.exit(1);
   662→   662→});
   663→   663→
   664→   664→// Graceful shutdown

   667→   667→  try { await browser.close(); } catch (_) {}
   668→   668→  httpServer.close();
   669→   669→  process.exit(0);
   670→   670→};

   672→   672→process.on('SIGTERM', () => shutdown('SIGTERM'));
