     1→// scripts/manual-login.js
     2→// CapCut manual login — works on HEADLESS servers (no X server / no display).
     3→//
     4→// Strategi:
     5→// 1. Launch Chromium dalam headless mode (modern headless, bukan old headless).
     6→// 2. Buka https://www.capcut.com/login — CapCut akan tampilkan QR code.
     7→// 3. Screenshot QR code berkala ke tmp/qr-latest.png.
     8→// 4. Jalankan HTTP server kecil di port 3001 (CAPCUT_LOGIN_PORT) untuk serve screenshot.
     9→// 5. User buka SSH tunnel: ssh -L 3001:localhost:3001 root@<server> -p <port>
    10→//    lalu buka http://localhost:3001/ di browser lokal → scan QR pake HP.
    11→// 6. Script poll untuk detect login (cookies, URL change, avatar element).
    12→// 7. Setelah login terdeteksi → session disimpan ke .capcut-profile/ (userDataDir).
    13→//
    14→// Cara pakai di server headless:
    15→//   npm run login:manual
    16→//   # di terminal lokal (laptop): ssh -L 3001:localhost:3001 root@sakura.proxy.rlwy.net -p 39551
    17→//   # buka http://localhost:3001/ di browser lokal → scan QR pake CapCut HP
    18→//
    19→// Cara pakai di local dengan display (desktop):
    20→//   DISPLAY=:0 npm run login:manual   # otomatis pakai headless:false

   254→async function findMobileLoginButton() {
   255→  // Strategy 1: find clickable element by text content
   256→  const result = await page.evaluate((texts) => {
   257→    const lowerTexts = texts.map((t) => t.toLowerCase());
   258→    const clickableSelectors = 'button, a, div[role="button"], span[role="button"], [class*="btn" i], [class*="button" i]';
   259→
   260→    const candidates = Array.from(document.querySelectorAll(clickableSelectors));
   261→    for (const el of candidates) {
   262→      // Get text from element AND its children, trimmed
   263→      const text = (el.innerText || el.textContent || '').trim().toLowerCase();
   264→      if (!text || text.length > 100) continue; // skip huge elements / empty
   265→
   266→      const matches = lowerTexts.some((t) => text === t || text.includes(t));
   267→      if (!matches) continue;
   268→
   269→      // Skip elements with very small size (icons, hidden)
   270→      const rect = el.getBoundingClientRect();
   271→      if (rect.width < 100 || rect.height < 20) continue;
   272→      if (rect.top < 0 || rect.left < 0) continue;
   273→
   274→      // Skip if text contains "continue with google" or other non-mobile
   275→      if (text.includes('google') || text.includes('email') || text.includes('tiktok') || text.includes('facebook')) {
   276→        continue;
   277→      }
   278→
   279→      // Found it! Return selector info
   280→      el.setAttribute('data-capcut-mobile-btn', 'true');
   281→      return {
   282→        found: true,
   283→        text: text.slice(0, 80),
   284→        tag: el.tagName.toLowerCase(),
   285→        width: rect.width,
   286→        height: rect.height,
   287→        top: rect.top,
   288→        left: rect.left,
   289→      };
   290→    }
   291→    return { found: false };
   292→  }, MOBILE_BTN_TEXTS);
   293→
   294→  return result;
   295→}

   297→async function findQrCanvasOrImg() {
   298→  // Look for QR code element in modal/popup using multiple strategies
   299→  return await page.evaluate(() => {
   300→    // Strategy 1: canvas with reasonable QR dimensions (200-400px square)
   301→    const canvases = Array.from(document.querySelectorAll('canvas'));
   302→    for (const c of canvases) {
   303→      const rect = c.getBoundingClientRect();
   304→      // QR codes are usually square, 100-400px
   305→      if (Math.abs(rect.width - rect.height) < 20 && rect.width >= 100 && rect.width <= 500) {
   306→        // Skip if it's the CapCut logo or something
   307→        const parent = c.parentElement;
   308→        const parentClass = parent ? parent.className.toLowerCase() : '';
   309→        if (parentClass.includes('logo') || parentClass.includes('icon')) continue;
   310→        return {
   311→          found: true,
   312→          type: 'canvas',
   313→          width: rect.width,
   314→          height: rect.height,
   315→          top: rect.top,
   316→          left: rect.left,
   317→          parentClass: parentClass.slice(0, 80),
   318→        };
   319→      }
   320→    }
   321→
   322→    // Strategy 2: img with qr-ish src or alt
   323→    const imgs = Array.from(document.querySelectorAll('img'));
   324→    for (const i of imgs) {
   325→      const src = (i.src || '').toLowerCase();
   326→      const alt = (i.alt || '').toLowerCase();
   327→      const rect = i.getBoundingClientRect();
   328→      if (rect.width < 80) continue;
   329→      if (src.includes('qr') || alt.includes('qr') ||
   330→          (src.startsWith('data:image/png;base64,') && rect.width >= 100 && Math.abs(rect.width - rect.height) < 30)) {
   331→        return {
   332→          found: true,
   333→          type: 'img',
   334→          width: rect.width,
   335→          height: rect.height,
   336→          top: rect.top,
   337→          left: rect.left,
   338→          src: src.slice(0, 80),
   339→        };
   340→      }
   341→    }
   342→
   343→    // Strategy 3: SVG with class containing qr
   344→    const svgs = Array.from(document.querySelectorAll('svg'));
   345→    for (const s of svgs) {
   346→      const cls = (s.className?.baseVal || s.getAttribute('class') || '').toLowerCase();
   347→      const rect = s.getBoundingClientRect();
   348→      if (cls.includes('qr') && rect.width >= 100) {
   349→        return {
   350→          found: true,
   351→          type: 'svg',
   352→          width: rect.width,
   353→          height: rect.height,
   354→          top: rect.top,
   355→          left: rect.left,
   356→          cls: cls.slice(0, 80),
   357→        };
   358→      }
   359→    }
   360→
   361→    // Strategy 4: any element with qr in class name that's reasonably sized
   362→    const qrEls = Array.from(document.querySelectorAll('[class*="qr" i], [class*="qrcode" i], [class*="scan" i]'));
   363→    for (const el of qrEls) {
   364→      const rect = el.getBoundingClientRect();
   365→      const text = (el.innerText || el.textContent || '').toLowerCase();
   366→      // Skip "Continue with CapCut Mobile" button (it has qr icon class)
   367→      if (text.includes('continue with')) continue;
   368→      if (rect.width < 100 || rect.height < 100) continue;
   369→      return {
   370→        found: true,
   371→        type: 'element',
   372→        width: rect.width,
   373→        height: rect.height,
   374→        top: rect.top,
   375→        left: rect.left,
   376→        cls: (el.className || '').toString().slice(0, 80),
   377→      };
   378→    }
   379→
   380→    return { found: false };
   381→  });
   382→}

   395→      const centerY = mobileBtnInfo.top + mobileBtnInfo.height / 2;
   396→
   397→      // Use page.mouse for a REAL trusted mouse click (not JS synthetic event).
   398→      // React/Vue SPA frameworks often ignore dispatchEvent() — they need real
   399→      // trusted events with proper mouse move → down → up sequence.
   400→      try {
   401→        // Move mouse to button first (some SPA frameworks need mouseover)
   402→        await page.mouse.move(centerX, centerY, { steps: 5 });
   403→        await new Promise((r) => setTimeout(r, 200));
   404→        // Real click via Puppeteer mouse (sends trusted OS-level event)
   405→        await page.mouse.click(centerX, centerY, { delay: 50 });

   407→        logger.info({ x: centerX, y: centerY }, 'Real mouse click sent to button center');
   408→      } catch (e) {
   409→        logger.warn({ err: e.message }, 'Mouse click failed, trying element.click() fallback');
   410→        // Fallback: use elementHandle.click() (Puppeteer's native click)

   496→      frames: allFrames.map((f) => ({ url: f.url().slice(0, 120) })),
   497→    },
   498→    'All frames (main + iframes)'
   499→  );
   500→
   501→  // Check all open pages/tabs (maybe QR opened in new tab)
   502→  const allPages = await browser.pages();
   503→  logger.info({ pageCount: allPages.length, urls: allPages.map((p) => p.url().slice(0, 80)) }, 'All open browser pages');
   504→
   505→  // If popup detected, wait for it to fully load and screenshot it
   506→  if (popupPage) {
   507→    logger.info({ popupUrl: popupPage.url() }, 'Popup page detected, waiting for it to load...');
   508→    try {
   509→      await popupPage.waitForLoadState?.({ waitUntil: 'load', timeout: 10000 }).catch(() => {});
   510→      await new Promise((r) => setTimeout(r, 3000));
   511→      const popupUrl = popupPage.url();
   512→      logger.info({ popupUrl }, 'Popup page URL after wait');
   513→      // Take screenshot of popup page directly
   514→      const popupShotPath = path.join(screenshotDir, 'popup-qr.png');
   515→      await popupPage.screenshot({ path: popupShotPath, fullPage: false }).catch((e) => {
   516→        logger.warn({ err: e.message }, 'Popup screenshot failed');
   517→      });
   518→      // Also save as qr-latest.png so HTTP viewer shows it
   519→      await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   520→      logger.info({ popupShotPath, qrLatestPath }, 'Popup screenshot saved!');
   521→    } catch (e) {
   522→      logger.warn({ err: e.message }, 'Error while waiting for popup');
   523→    }
   524→  }
   525→}
   526→
   527→// ====== Polling: detect login + capture QR ======
   528→const startTime = Date.now();
   529→let lastScreenshotTime = 0;
   530→let qrFound = false;
   531→let tries = 0;
   532→
   533→async function checkLogin() {
   534→  tries++;
   535→  const elapsedMs = Date.now() - startTime;
   536→
   537→  if (elapsedMs > MAX_WAIT_MS) {
   538→    logger.warn({ elapsedMs }, `Timeout ${MAX_WAIT_MS / 1000}s. No login detected.`);
   539→    return 'timeout';
   540→  }
   541→
   542→  try {
   543→    // Use popupPage if it exists (where QR code lives), otherwise main page
   544→    const activePage = popupPage && !popupPage.isClosed() ? popupPage : page;
   545→    const currentUrl = activePage.url();
   546→
   547→    // Cek 1: URL berubah dari /login atau /scan-qr-code (redirect ke home/dashboard)
   548→    const stillOnLogin = /\/login|\/scan-qr-code/i.test(currentUrl);
   549→
   550→    // Cek 2: cookies session (check both pages, cookies are shared per profile)
   551→    const cookies = await page.cookies();
   552→    const sessionCookie = cookies.find(
   553→      (c) =>
   554→        /session|token|uid|passport|login|sid|tt_csrf|s_v_web_id/i.test(c.name) && c.value && c.value.length > 10
   555→    );
   556→
   557→    // Cek 3: avatar / user element di header (check main page)
   558→    let avatarFound = false;
   559→    if (!stillOnLogin) {
   560→      try {
   561→        avatarFound = await page.$(
   562→          '[class*="avatar" i], [data-e2e*="user" i], [data-testid*="user" i], [class*="user-info" i], [class*="header-user" i], [class*="account" i]'
   563→        ) ? true : false;
   564→      } catch (_) {}
   565→    }
   566→
   567→    if (sessionCookie && (!stillOnLogin || avatarFound)) {
   568→      logger.info({ url: currentUrl, cookie: sessionCookie?.name, avatarFound }, 'LOGIN SUCCESS detected!');
   569→      return 'success';
   570→    }
   571→
   572→    // Screenshot QR code — prefer popupPage (where QR actually is)
   573→    const now = Date.now();
   574→    if (now - lastScreenshotTime > 1500) {
   575→      try {
   576→        if (popupPage && !popupPage.isClosed()) {
   577→          // Screenshot the popup page (QR code lives here!)
   578→          await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   579→          if (!qrFound) {
   580→            logger.info({ popupUrl: popupPage.url() }, 'Capturing QR from popup page');
   581→            qrFound = true;
   582→          }
   583→        } else {
   584→          // Fallback: try to find QR element on main page
   585→          const found = await findQrCanvasOrImg();
   586→          if (found && found.found) {
   587→            if (!qrFound) {
   588→              logger.info(
   589→                { type: found.type, w: Math.round(found.width), h: Math.round(found.height) },
   590→                'QR code element detected on main page, capturing...'
   591→              );
   592→              qrFound = true;
   593→            }
   594→            await page.screenshot({
   595→              path: qrLatestPath,
   596→              clip: {
   597→                x: Math.max(0, found.left - 30),
   598→                y: Math.max(0, found.top - 30),
   599→                width: found.width + 60,
   600→                height: found.height + 60,
   601→              },
   602→            });
   603→          } else {
   604→            await page.screenshot({ path: qrLatestPath, fullPage: false });
   605→          }
   606→        }
   607→        lastScreenshotTime = now;
   608→
   609→        // Save periodic debug screenshot
   610→        if (tries % 5 === 0) {
   611→          const debugPath = path.join(screenshotDir, `debug-${tries}.png`);
   612→          await activePage.screenshot({ path: debugPath, fullPage: false }).catch(() => {});
   613→        }
   614→      } catch (e) {
   615→        logger.warn({ err: e.message }, 'Screenshot error');
   616→      }
   617→    }
   618→
   619→    if (tries % 6 === 0) {
   620→      logger.info(
   621→        { tries, elapsedS: Math.round(elapsedMs / 1000), url: currentUrl, qrFound, hasSessionCookie: !!sessionCookie, mobileBtnClicked, popupAlive: popupPage && !popupPage.isClosed() },
   622→        'Waiting for login...'
   623→      );
   624→    }
   625→  } catch (e) {
   626→    logger.warn({ err: e.message }, 'Polling error');
   627→  }
   628→
   629→  return 'continue';
   630→}
   631→
   632→const pollLoop = async () => {
   633→  while (true) {
   634→    const result = await checkLogin();
   635→    if (result === 'success') {
   636→      // Beri delay sedikit untuk pastikan session tersimpan
   637→      await new Promise((r) => setTimeout(r, 2000));
   638→      try {
   639→        await browser.close();
   640→      } catch (_) {}
   641→      httpServer.close();
   642→      logger.info(`✓ Session saved to ${userDataDir}`);
   643→      logger.info('Set CAPCUT_USER_DATA_DIR=./.capcut-profile di .env untuk reuse session ini.');
   644→      process.exit(0);
   645→    }
   646→    if (result === 'timeout') {
   647→      try {
   648→        await browser.close();
   649→      } catch (_) {}
   650→      httpServer.close();
   651→      process.exit(1);
   652→    }
   653→    await new Promise((r) => setTimeout(r, POLL_INTERVAL_MS));
   654→  }
   655→};
   656→
   657→pollLoop().catch((e) => {
   658→  logger.error({ err: e.message }, 'Fatal error in poll loop');
   659→  try { browser.close(); } catch (_) {}
   660→  httpServer.close();
   661→  process.exit(1);
   662→});
   663→
   664→// Graceful shutdown
   665→const shutdown = async (sig) => {
   666→  logger.info({ sig }, 'Received signal, closing browser...');
   667→  try { await browser.close(); } catch (_) {}
   668→  httpServer.close();
   669→  process.exit(0);
   670→};
   671→process.on('SIGINT', () => shutdown('SIGINT'));
   672→process.on('SIGTERM', () => shutdown('SIGTERM'));
