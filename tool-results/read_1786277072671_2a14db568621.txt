     1→// scripts/manual-login.js
     2→// CapCut manual login — works on HEADLESS servers (no X server / no display).
     3→//
     4→// Strategi:
     5→// 1. Launch Chromium dalam headless mode (modern headless, bukan old headless).
     6→// 2. Buka https://www.capcut.com/login — CapCut akan tampilkan QR code.
     7→// 3. Screenshot QR code berkala ke tmp/qr-latest.png.
     8→// 4. Jalankan HTTP server kecil di port 3001 (CAPCUT_LOGIN_PORT) untuk serve screenshot.
     9→// 5. User buka SSH tunnel: ssh -L 3001:localhost:3001 root@<server> -p <port>
    10→//    lalu buka http://localhost:3001/ di browser lokal → scan QR pake HP.
    11→// 6. Script poll untuk detect login (cookies, URL change, avatar element).
    12→// 7. Setelah login terdeteksi → session disimpan ke .capcut-profile/ (userDataDir).
    13→//
    14→// Cara pakai di server headless:
    15→//   npm run login:manual
    16→//   # di terminal lokal (laptop): ssh -L 3001:localhost:3001 root@sakura.proxy.rlwy.net -p 39551
    17→//   # buka http://localhost:3001/ di browser lokal → scan QR pake CapCut HP
    18→//
    19→// Cara pakai di local dengan display (desktop):

    59→function isPortFree(port) {
    60→  return new Promise((resolve) => {
    61→    const tester = http.createServer();

    63→    tester.once('listening', () => {

    70→// Helper: cari port kosong mulai dari preferred, naik 1 per 1 sampai preferred+9
    71→async function findFreePort(preferred) {
    72→  for (let p = preferred; p <= preferred + 9; p++) {
    73→    if (await isPortFree(p)) return p;
    74→  }
    75→  return null; // semua sibuk
    76→}

    83→  );

   292→async function findMobileLoginButton() {
   293→  // Strategy 1: find clickable element by text content
   294→  const result = await page.evaluate((texts) => {
   295→    const lowerTexts = texts.map((t) => t.toLowerCase());
   296→    const clickableSelectors = 'button, a, div[role="button"], span[role="button"], [class*="btn" i], [class*="button" i]';
   297→
   298→    const candidates = Array.from(document.querySelectorAll(clickableSelectors));
   299→    for (const el of candidates) {
   300→      // Get text from element AND its children, trimmed
   301→      const text = (el.innerText || el.textContent || '').trim().toLowerCase();
   302→      if (!text || text.length > 100) continue; // skip huge elements / empty
   303→
   304→      const matches = lowerTexts.some((t) => text === t || text.includes(t));
   305→      if (!matches) continue;
   306→
   307→      // Skip elements with very small size (icons, hidden)
   308→      const rect = el.getBoundingClientRect();
   309→      if (rect.width < 100 || rect.height < 20) continue;
   310→      if (rect.top < 0 || rect.left < 0) continue;
   311→
   312→      // Skip if text contains "continue with google" or other non-mobile
   313→      if (text.includes('google') || text.includes('email') || text.includes('tiktok') || text.includes('facebook')) {
   314→        continue;
   315→      }
   316→
   317→      // Found it! Return selector info
   318→      el.setAttribute('data-capcut-mobile-btn', 'true');
   319→      return {
   320→        found: true,
   321→        text: text.slice(0, 80),
   322→        tag: el.tagName.toLowerCase(),
   323→        width: rect.width,
   324→        height: rect.height,
   325→        top: rect.top,
   326→        left: rect.left,
   327→      };
   328→    }
   329→    return { found: false };
   330→  }, MOBILE_BTN_TEXTS);
   331→
   332→  return result;
   333→}

   335→async function findQrCanvasOrImg() {
   336→  // Look for QR code element in modal/popup using multiple strategies
   337→  return await page.evaluate(() => {
   338→    // Strategy 1: canvas with reasonable QR dimensions (200-400px square)
   339→    const canvases = Array.from(document.querySelectorAll('canvas'));
   340→    for (const c of canvases) {
   341→      const rect = c.getBoundingClientRect();
   342→      // QR codes are usually square, 100-400px
   343→      if (Math.abs(rect.width - rect.height) < 20 && rect.width >= 100 && rect.width <= 500) {
   344→        // Skip if it's the CapCut logo or something
   345→        const parent = c.parentElement;
   346→        const parentClass = parent ? parent.className.toLowerCase() : '';
   347→        if (parentClass.includes('logo') || parentClass.includes('icon')) continue;
   348→        return {
   349→          found: true,
   350→          type: 'canvas',
   351→          width: rect.width,
   352→          height: rect.height,
   353→          top: rect.top,
   354→          left: rect.left,
   355→          parentClass: parentClass.slice(0, 80),
   356→        };
   357→      }
   358→    }
   359→
   360→    // Strategy 2: img with qr-ish src or alt
   361→    const imgs = Array.from(document.querySelectorAll('img'));
   362→    for (const i of imgs) {
   363→      const src = (i.src || '').toLowerCase();
   364→      const alt = (i.alt || '').toLowerCase();
   365→      const rect = i.getBoundingClientRect();
   366→      if (rect.width < 80) continue;
   367→      if (src.includes('qr') || alt.includes('qr') ||
   368→          (src.startsWith('data:image/png;base64,') && rect.width >= 100 && Math.abs(rect.width - rect.height) < 30)) {
   369→        return {
   370→          found: true,
   371→          type: 'img',
   372→          width: rect.width,
   373→          height: rect.height,
   374→          top: rect.top,
   375→          left: rect.left,
   376→          src: src.slice(0, 80),
   377→        };
   378→      }
   379→    }
   380→
   381→    // Strategy 3: SVG with class containing qr
   382→    const svgs = Array.from(document.querySelectorAll('svg'));
   383→    for (const s of svgs) {
   384→      const cls = (s.className?.baseVal || s.getAttribute('class') || '').toLowerCase();
   385→      const rect = s.getBoundingClientRect();
   386→      if (cls.includes('qr') && rect.width >= 100) {
   387→        return {
   388→          found: true,
   389→          type: 'svg',
   390→          width: rect.width,
   391→          height: rect.height,
   392→          top: rect.top,
   393→          left: rect.left,
   394→          cls: cls.slice(0, 80),
   395→        };
   396→      }
   397→    }
   398→
   399→    // Strategy 4: any element with qr in class name that's reasonably sized
   400→    const qrEls = Array.from(document.querySelectorAll('[class*="qr" i], [class*="qrcode" i], [class*="scan" i]'));
   401→    for (const el of qrEls) {
   402→      const rect = el.getBoundingClientRect();
   403→      const text = (el.innerText || el.textContent || '').toLowerCase();
   404→      // Skip "Continue with CapCut Mobile" button (it has qr icon class)
   405→      if (text.includes('continue with')) continue;
   406→      if (rect.width < 100 || rect.height < 100) continue;
   407→      return {
   408→        found: true,
   409→        type: 'element',
   410→        width: rect.width,
   411→        height: rect.height,
   412→        top: rect.top,
   413→        left: rect.left,
   414→        cls: (el.className || '').toString().slice(0, 80),
   415→      };
   416→    }
   417→
   418→    return { found: false };
   419→  });
   420→}

   485→  // Diagnostic: check what changed after click
   486→  const diag = await page.evaluate(() => {

   489→    const iframes = document.querySelectorAll('iframe').length;

   495→        const rect = d.getBoundingClientRect();
   496→        return (
   497→          (s.position === 'fixed' || s.position === 'absolute') &&
   498→          rect.width > 200 &&
   499→          rect.height > 200 &&
   500→          s.zIndex !== 'auto' &&
   501→          parseInt(s.zIndex || '0', 10) > 100
   502→        );
   503→      }).length;
   504→    return { allPages, canvases, iframes, dialogs, overlays, url: location.href };
   505→  }).catch(() => ({}));
   506→  logger.info({ diag }, 'Post-click diagnostics (canvases, iframes, dialogs, overlays)');
   507→
   508→  // List all iframes with their src + position (CapCut Mobile login uses an iframe!)
   509→  const iframeInfos = await page.evaluate(() => {
   510→    return Array.from(document.querySelectorAll('iframe')).map((f) => {
   511→      const r = f.getBoundingClientRect();
   512→      return {
   513→        src: (f.src || '').slice(0, 150),
   514→        width: r.width,
   515→        height: r.height,
   516→        top: r.top,
   517→        left: r.left,
   518→        visible: r.width > 0 && r.height > 0,
   519→      };
   520→    });
   521→  }).catch(() => []);
   522→  logger.info({ iframeCount: iframeInfos.length, iframes: iframeInfos }, 'Iframes on page');
   523→
   524→  // Take a screenshot to see if click worked
   525→  const afterClickPath = path.join(screenshotDir, 'after-click.png');
   526→  await page.screenshot({ path: afterClickPath, fullPage: false }).catch(() => {});
   527→  logger.info({ path: afterClickPath }, 'Screenshot after click saved');
   528→
   529→  // List all puppeteer frames (can access same-origin iframe content)
   530→  const allFrames = page.frames();
   531→  logger.info(
   532→    {
   533→      frameCount: allFrames.length,
   534→      frames: allFrames.map((f) => ({ url: f.url().slice(0, 120) })),
   535→    },
   536→    'All frames (main + iframes)'
   537→  );
   538→
   539→  // Check all open pages/tabs (maybe QR opened in new tab)
   540→  const allPages = await browser.pages();
   541→  logger.info({ pageCount: allPages.length, urls: allPages.map((p) => p.url().slice(0, 80)) }, 'All open browser pages');
   542→
   543→  // If popup detected, wait for it to fully load and screenshot it
   544→  if (popupPage) {
   545→    logger.info({ popupUrl: popupPage.url() }, 'Popup page detected, waiting for it to load...');
   546→    try {
   547→      await popupPage.waitForLoadState?.({ waitUntil: 'load', timeout: 10000 }).catch(() => {});
   548→      await new Promise((r) => setTimeout(r, 3000));
   549→      const popupUrl = popupPage.url();
   550→      logger.info({ popupUrl }, 'Popup page URL after wait');
   551→      // Take screenshot of popup page directly
   552→      const popupShotPath = path.join(screenshotDir, 'popup-qr.png');
   553→      await popupPage.screenshot({ path: popupShotPath, fullPage: false }).catch((e) => {
   554→        logger.warn({ err: e.message }, 'Popup screenshot failed');
   555→      });
   556→      // Also save as qr-latest.png so HTTP viewer shows it
   557→      await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   558→      logger.info({ popupShotPath, qrLatestPath }, 'Popup screenshot saved!');
   559→    } catch (e) {
   560→      logger.warn({ err: e.message }, 'Error while waiting for popup');
   561→    }
   562→  }
   563→}
   564→
   565→// ====== Polling: detect login + capture QR ======
   566→const startTime = Date.now();
   567→let lastScreenshotTime = 0;
   568→let qrFound = false;
   569→let tries = 0;
   570→
   571→async function checkLogin() {
   572→  tries++;
   573→  const elapsedMs = Date.now() - startTime;
   574→
   575→  if (elapsedMs > MAX_WAIT_MS) {
   576→    logger.warn({ elapsedMs }, `Timeout ${MAX_WAIT_MS / 1000}s. No login detected.`);
   577→    return 'timeout';
   578→  }
   579→
   580→  try {
   581→    // Use popupPage if it exists (where QR code lives), otherwise main page
   582→    const activePage = popupPage && !popupPage.isClosed() ? popupPage : page;
   583→    const currentUrl = activePage.url();
   584→
   585→    // Cek 1: URL berubah dari /login atau /scan-qr-code (redirect ke home/dashboard)
   586→    const stillOnLogin = /\/login|\/scan-qr-code/i.test(currentUrl);
   587→
   588→    // Cek 2: cookies session (check both pages, cookies are shared per profile)
   589→    const cookies = await page.cookies();
   590→    const sessionCookie = cookies.find(
   591→      (c) =>
   592→        /session|token|uid|passport|login|sid|tt_csrf|s_v_web_id/i.test(c.name) && c.value && c.value.length > 10
   593→    );
   594→
   595→    // Cek 3: avatar / user element di header (check main page)
   596→    let avatarFound = false;
   597→    if (!stillOnLogin) {
   598→      try {
   599→        avatarFound = await page.$(
   600→          '[class*="avatar" i], [data-e2e*="user" i], [data-testid*="user" i], [class*="user-info" i], [class*="header-user" i], [class*="account" i]'
   601→        ) ? true : false;
   602→      } catch (_) {}
   603→    }
   604→
   605→    if (sessionCookie && (!stillOnLogin || avatarFound)) {
   606→      logger.info({ url: currentUrl, cookie: sessionCookie?.name, avatarFound }, 'LOGIN SUCCESS detected!');
   607→      return 'success';
   608→    }
   609→
   610→    // Screenshot QR code — prefer popupPage (where QR actually is)
   611→    const now = Date.now();
   612→    if (now - lastScreenshotTime > 1500) {
   613→      try {
   614→        if (popupPage && !popupPage.isClosed()) {
   615→          // Screenshot the popup page (QR code lives here!)
   616→          await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   617→          if (!qrFound) {
   618→            logger.info({ popupUrl: popupPage.url() }, 'Capturing QR from popup page');
   619→            qrFound = true;
   620→          }
   621→        } else {
   622→          // Fallback: try to find QR element on main page
   623→          const found = await findQrCanvasOrImg();
   624→          if (found && found.found) {
   625→            if (!qrFound) {
   626→              logger.info(
   627→                { type: found.type, w: Math.round(found.width), h: Math.round(found.height) },
   628→                'QR code element detected on main page, capturing...'
   629→              );
   630→              qrFound = true;
   631→            }
   632→            await page.screenshot({
   633→              path: qrLatestPath,
   634→              clip: {
   635→                x: Math.max(0, found.left - 30),
   636→                y: Math.max(0, found.top - 30),
   637→                width: found.width + 60,
   638→                height: found.height + 60,
   639→              },
   640→            });
   641→          } else {
   642→            await page.screenshot({ path: qrLatestPath, fullPage: false });
   643→          }
   644→        }
   645→        lastScreenshotTime = now;
   646→
   647→        // Save periodic debug screenshot
   648→        if (tries % 5 === 0) {
   649→          const debugPath = path.join(screenshotDir, `debug-${tries}.png`);
   650→          await activePage.screenshot({ path: debugPath, fullPage: false }).catch(() => {});
   651→        }
   652→      } catch (e) {
   653→        logger.warn({ err: e.message }, 'Screenshot error');
   654→      }
   655→    }
   656→
   657→    if (tries % 6 === 0) {
   658→      logger.info(
   659→        { tries, elapsedS: Math.round(elapsedMs / 1000), url: currentUrl, qrFound, hasSessionCookie: !!sessionCookie, mobileBtnClicked, popupAlive: popupPage && !popupPage.isClosed() },
   660→        'Waiting for login...'
   661→      );
   662→    }
   663→  } catch (e) {
   664→    logger.warn({ err: e.message }, 'Polling error');
   665→  }
   666→
   667→  return 'continue';
   668→}
   669→
   670→const pollLoop = async () => {
   671→  while (true) {
   672→    const result = await checkLogin();
   673→    if (result === 'success') {
   674→      // Beri delay sedikit untuk pastikan session tersimpan
   675→      await new Promise((r) => setTimeout(r, 2000));
   676→      try {
   677→        await browser.close();
   678→      } catch (_) {}
   679→      httpServer.close();
   680→      logger.info(`✓ Session saved to ${userDataDir}`);
   681→      logger.info('Set CAPCUT_USER_DATA_DIR=./.capcut-profile di .env untuk reuse session ini.');
   682→      process.exit(0);
   683→    }
   684→    if (result === 'timeout') {
   685→      try {
   686→        await browser.close();
   687→      } catch (_) {}
   688→      httpServer.close();
   689→      process.exit(1);
   690→    }
   691→    await new Promise((r) => setTimeout(r, POLL_INTERVAL_MS));
   692→  }
   693→};
   694→
   695→pollLoop().catch((e) => {
   696→  logger.error({ err: e.message }, 'Fatal error in poll loop');
   697→  try { browser.close(); } catch (_) {}
   698→  httpServer.close();
   699→  process.exit(1);
   700→});
   701→
   702→// Graceful shutdown
   703→const shutdown = async (sig) => {
   704→  logger.info({ sig }, 'Received signal, closing browser...');
   705→  try { await browser.close(); } catch (_) {}
   706→  httpServer.close();
   707→  process.exit(0);
   708→};
   709→process.on('SIGINT', () => shutdown('SIGINT'));
   710→process.on('SIGTERM', () => shutdown('SIGTERM'));
