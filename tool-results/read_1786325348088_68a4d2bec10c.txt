    14→//   8. downloadVideo()      — stream MP4 to disk

    16→// Auth: cookies from .capcut-profile (logged-in via manual-login.js).
    17→// Required cookies: passport_csrf_token, sessionid, passport, ttwid.

    39→/**
    40→ * CapCut request signature.
    41→ * Algorithm (reverse-engineered from editor bundle-018.js + bundle-035.js):
    42→ *
    43→ *   u = last 7 chars of URL path (or full path if shorter)
    44→ *   sign = md5(`9e2c|${u}|${pf}|${appvr}|${deviceTime}|${tdid}|11ac`).toLowerCase()
    45→ *
    46→ * Verified against real captured request:
    47→ *   path: /cc/v1/workspace/get_user_workspaces
    48→ *   pf=7, appvr=5.8.0, device-time=1786317731, tdid=""
    49→ *   => sign = 05b31d04820ddeb77bad9a583a34f79d ✓
    50→ *
    51→ * @param {string} urlPath - URL pathname (e.g., /cc/v1/workspace/get_user_workspaces)
    52→ * @param {Object} opts - { pf, appvr, tdid, deviceTime }
    53→ * @returns {string} 32-char MD5 hex
    54→ */
    55→function calcSign(urlPath, { pf = '7', appvr = '15.4.0', tdid = 'web', deviceTime } = {}) {
    56→  const u = urlPath.length >= 7 ? urlPath.slice(-7) : urlPath;
    57→  const ts = deviceTime || Math.floor(Date.now() / 1000);
    58→  const payload = `9e2c|${u}|${pf}|${appvr}|${ts}|${tdid}|11ac`;
    59→  return crypto.createHash('md5').update(payload).digest('hex');
    60→}

    91→/**
    92→ * Load cookies from Chrome profile (SQLite).
    93→ * Falls back to puppeteer extraction if direct read fails.
    94→ * @returns {Promise<string>} cookie header string
    95→ */
    96→async function loadCookieHeader() {
    97→  // Lazy-load puppeteer to read cookies (avoids Chrome SQLite encryption complexity)
    98→  const { default: puppeteer } = await import('puppeteer');
    99→  const userDataDir = process.env.CAPCUT_USER_DATA_DIR || path.join(projectRoot, '.capcut-profile');
   100→
   101→  // Cleanup stale locks
   102→  for (const lock of ['SingletonLock', 'SingletonCookie', 'SingletonSocket']) {
   103→    try { fs.rmSync(path.join(userDataDir, lock), { force: true }); } catch {}
   104→  }
   105→
   106→  const browser = await puppeteer.launch({
   107→    headless: 'new',
   108→    userDataDir,
   109→    args: ['--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage'],
   110→  });
   111→  try {
   112→    const page = await browser.newPage();
   113→    await page.goto(`${HOSTS.WEB}/`, { waitUntil: 'domcontentloaded', timeout: 30000 });
   114→    const cookies = await browser.cookies(HOSTS.WEB, 'https://capcut.com');
   115→    if (cookies.length === 0) {
   116→      throw new Error('No cookies found in profile. Run npm run login:manual first.');
   117→    }
   118→    const header = cookies.map(c => `${c.name}=${c.value}`).join('; ');
   119→    logger.info({ cookieCount: cookies.length }, 'Loaded cookies from profile');
   120→    return header;
   121→  } finally {
   122→    await browser.close();
   123→  }
   124→}

   137→  async _init() {
   138→    if (this._axios) return;
   139→    this.cookieHeader = await loadCookieHeader();
   140→    this._axios = axios.create({
   141→      headers: { ...DEFAULT_HEADERS, Cookie: this.cookieHeader },
   142→      timeout: 30000,
   143→      validateStatus: s => s < 500, // don't throw on 4xx — handle gracefully
   144→    });
   145→
   146→    // Add sign interceptor — computes sign for every request based on URL path.
   147→    // Algorithm: md5(`9e2c|${last7_of_path}|${pf}|${appvr}|${deviceTime}|${tdid}|11ac`)
   148→    this._axios.interceptors.request.use((config) => {
   149→      const url = config.url || '';
   150→      let urlPath;
   151→      try {
   152→        urlPath = new URL(url, config.baseURL || 'https://edit-api-sg.capcut.com').pathname;
   153→      } catch {
   154→        urlPath = url;
   155→      }
   156→      const deviceTime = Math.floor(Date.now() / 1000);
   157→      const pf = config.headers?.pf || DEFAULT_HEADERS.pf;
   158→      const appvr = config.headers?.appvr || DEFAULT_HEADERS.appvr;
   159→      const tdid = config.headers?.tdid ?? DEFAULT_HEADERS.tdid;
   160→
   161→      config.headers = {
   162→        ...config.headers,
   163→        'device-time': String(deviceTime),
   164→        'sign': calcSign(urlPath, { pf, appvr, tdid, deviceTime }),
   165→      };
   166→      return config;
   167→    });
   168→
   169→    logger.info({ workspaceId: this.workspaceId }, 'CapCutDirectAPI initialized (with sign interceptor)');
   170→  }

   218→  /**

   223→   *   - template_data: the draft JSON serialized as a string
   224→   *   - template_meta: JSON.stringify({draft, uploadSource, createSource})

   386→  /**
   387→   * Step 4: Patch draft materials — swap template materials with user assets.
   388→   * This is template-specific; implementation depends on draft content structure.
   389→   *
   390→   * @param {Object} draftContent - parsed draft content from getTemplateFile
   391→   * @param {Array<{asset_id, video_id, file_url}>} userAssets - uploaded assets
   392→   * @param {Object} opts - { materialIndices?: number[] } which materials to replace (default: all image materials)
   393→   * @returns {Object} patched draft content
   394→   */
   395→  patchDraftMaterials(draftContent, userAssets, { materialIndices } = {}) {
   396→    // CapCut draft content typically has:
   397→    //   content.materials.videos[]  — video material slots
   398→    //   content.materials.images[]  — image material slots (what we want to replace)
   399→    //   content.materials.audios[]
   400→    //   content.tracks[]            — timeline tracks referencing materials by ID
   401→    //
   402→    // For templates with N image slots, we replace materials.images[i] with userAssets[i].
   403→
   404→    if (!draftContent || typeof draftContent !== 'object') {
   405→      throw new Error('Invalid draft content');
   406→    }
   407→
   408→    // Deep clone to avoid mutating input
   409→    const patched = JSON.parse(JSON.stringify(draftContent));
   410→    const materials = patched.materials || (patched.materials = {});
   411→    const images = materials.images || (materials.images = []);
   412→
   413→    logger.info(
   414→      { imageSlots: images.length, userAssets: userAssets.length, materialIndices },
   415→      'patchDraftMaterials'
   416→    );
   417→
   418→    // Determine which slots to replace
   419→    const indicesToReplace = materialIndices
   420→      ? materialIndices
   421→      : images.map((_, i) => i).slice(0, userAssets.length);
   422→
   423→    for (let i = 0; i < indicesToReplace.length && i < userAssets.length; i++) {
   424→      const idx = indicesToReplace[i];
   425→      const asset = userAssets[i];
   426→      if (idx >= images.length) {
   427→        logger.warn({ idx, imageCount: images.length }, 'patchDraftMaterials: index out of range, skipping');
   428→        continue;
   429→      }
   430→
   431→      // Preserve existing material ID & duration, swap the source URL/asset_id
   432→      const existing = images[idx] || {};
   433→      images[idx] = {
   434→        ...existing,
   435→        // Common fields across CapCut versions
   436→        asset_id: asset.asset_id || existing.asset_id,
   437→        video_id: asset.video_id || existing.video_id,
   438→        file_url: asset.file_url || asset.url || existing.file_url,
   439→        url: asset.file_url || asset.url || existing.url,
   440→        source: 'cloud',
   441→        // Keep existing material_id so tracks still reference correctly
   442→        material_id: existing.material_id,
   443→      };
   444→    }
   445→
   446→    return patched;
   447→  }

   449→  /**
   450→   * Step 6: Create render task.
   451→   *
   452→   * VERIFIED body schema (from bundle-018.js offset 96764 + live test on production):
   453→   *   - BOTH draft_id AND package_id are required for the task to be created.
   454→   *   - draft_id = the ORIGINAL package_key we generated in saveDraft()
   455→   *   - package_id = the package_id returned by saveDraft()
   456→   *   - NO submit_id field (that's for a different endpoint, createExportTask)
   457→   *   - If only draft_id or only package_id is provided, returns ret=1000 "param error".
   458→   *
   459→   * @param {Object} opts
   460→   *   - draftId (required) — the original package_key from saveDraft
   461→   *   - packageId (required) — the package_id returned by saveDraft
   462→   *   - videoName (default 'CapCut Render')
   463→   *   - definition (default '720p') — 480p/720p/1080p
   464→   *   - width, height, fps (defaults: 1080x1920 @ 30fps)
   465→   *   - format (default 'mp4')
   466→   *   - duration (default 10000, in ms)
   467→   * @returns {Promise<{task_id, miss_materials, raw}>}
   468→   */
   469→  async createRenderTask({
   470→    draftId,
   471→    packageId,
   472→    videoName = 'CapCut Render',
   473→    definition = '720p',
   474→    width = 1080,
   475→    height = 1920,
   476→    fps = 30,
   477→    format = 'mp4',
   478→    duration = 10000,
   479→    cover = '',
   480→  } = {}) {
   481→    await this._init();
   482→    if (!draftId) throw new Error('draftId required (use the original package_key from saveDraft)');
   483→    if (!packageId) throw new Error('packageId required (use the package_id returned by saveDraft)');
   484→
   485→    const body = {
   486→      app_version: '1.0.0.285',
   487→      sdk_version: '127.0.0',
   488→      extra: '{}',
   489→      type: 0,
   490→      region: KNOWN.REGION,
   491→      app_id: APP_ID,
   492→      width,
   493→      height,
   494→      fps,
   495→      format,
   496→      cover,
   497→      duration,
   498→      quality: 100,
   499→      definition,
   500→      task_id: '',
   501→      video_name: videoName,
   502→      draft_id: draftId,
   503→      package_id: packageId,
   504→      video_id: '',
   505→      video_path: '',
   506→      group_id: '',
   507→      custom_info: '{}',
   508→      from_workspace_id: this.workspaceId,
   509→      to_workspace_id: this.workspaceId,
   510→      force_export: false,
   511→    };
   512→
   513→    logger.info({ draftId, packageId, definition, videoName }, 'createRenderTask');
   514→    const res = await this._axios.post(
   515→      `${HOSTS.EDIT}/lv/v1/render_task/create`,
   516→      body
   517→    );
   518→    const data = this._unwrap(res.data);
   519→    if (String(data.task_id) === '0') {
   520→      throw new Error(`render_task/create returned task_id=0 (not created). ret=${res.data?.ret} errmsg=${res.data?.errmsg}`);
   521→    }
   522→    return {
   523→      task_id: data.task_id,
   524→      miss_materials: data.miss_materials,
   525→      raw: data,
   526→    };
   527→  }
   528→
   529→  /**
   530→   * Step 7: Poll render task status.
   531→   *
   532→   * VERIFIED response format (from live test on production):
   533→   *   The batch_get response data is a DICT keyed by task_id, e.g.:
   534→   *   { "7672189146483589121": { id, status, progress, video_url, ... } }
   535→   *
   536→   * Status values observed:
   537→   *   0  = waiting/queued
   538→   *   1  = processing (progress 1-99)
   539→   *   2  = success (video_url will be set)
   540→   *   -1 = failed (check render_ret_code / task_err_code)
   541→   *   4  = canceled
   542→   *
   543→   * @param {string} taskId
   544→   * @param {Object} opts - { intervalMs, timeoutMs, onProgress }
   545→   * @returns {Promise<{status, video_url, progress, raw}>}
   546→   */
   547→  async pollRenderTask(taskId, {
   548→    intervalMs = 3000,
   549→    timeoutMs = 300000,
   550→    onProgress = null,
   551→  } = {}) {
   552→    await this._init();
   553→    const start = Date.now();
   554→    while (Date.now() - start < timeoutMs) {
   555→      const res = await this._axios.post(
   556→        `${HOSTS.EDIT}/lv/v1/render_task/batch_get`,
   557→        { task_ids: [taskId] }
   558→      );
   559→      const data = this._unwrap(res.data);
   560→      // Response format variations:
   561→      //   - data[taskId] = {id, status, progress, video_url, ...} (dict keyed by task_id)
   562→      //   - data.render_task = {id, status, ...} (single task)
   563→      //   - data.render_task[taskId] = {...} (dict under render_task key)
   564→      //   - data.tasks[] = [...] (array)
   565→      //   - data.task_list[] = [...] (array)
   566→      //   - data itself = {status, id, ...} (already unwrapped)
   567→      let taskInfo = null;
   568→      if (data && typeof data === 'object' && data[taskId]) {
   569→        taskInfo = data[taskId];
   570→      } else if (data?.render_task) {
   571→        if (Array.isArray(data.render_task)) {
   572→          taskInfo = data.render_task[0];
   573→        } else if (data.render_task[taskId]) {
   574→          taskInfo = data.render_task[taskId];
   575→        } else {
   576→          taskInfo = data.render_task;
   577→        }
   578→      } else if (Array.isArray(data.tasks) && data.tasks.length > 0) {
   579→        taskInfo = data.tasks[0];
   580→      } else if (Array.isArray(data.task_list) && data.task_list.length > 0) {
   581→        taskInfo = data.task_list[0];
   582→      } else if (data && (data.status !== undefined || data.id)) {
   583→        taskInfo = data;
   584→      }
   585→
   586→      if (!taskInfo) {
   587→        logger.warn({ taskId, dataKeys: Object.keys(data || {}) }, 'pollRenderTask: no task info found');
   588→        await new Promise(r => setTimeout(r, intervalMs));
   589→        continue;
   590→      }
   591→
   592→      const status = taskInfo.status ?? taskInfo.state;
   593→      const progress = taskInfo.progress || 0;
   594→      const videoUrl = taskInfo.video_url || taskInfo.video?.url || taskInfo.download_url;
   595→
   596→      logger.info({ taskId, status, progress, hasUrl: !!videoUrl }, 'pollRenderTask');
   597→
   598→      if (onProgress) {
   599→        try { onProgress({ status, progress, videoUrl, raw: taskInfo }); } catch {}
   600→      }
   601→
   602→      // Status: 2=success, -1=failed, 4=canceled
   603→      if (status === 2 || status === 'success' || status === 'done' || status === 'completed') {
   604→        return {
   605→          status: 'completed',
   606→          progress: 100,
   607→          video_url: videoUrl,
   608→          raw: taskInfo,
   609→        };
   610→      }
   611→      if (status === -1 || status === 3 || status === 'failed' || status === 'error') {
   612→        const errMsg = `Render failed: status=${status} ret_code=${taskInfo.render_ret_code || ''} err_code=${taskInfo.task_err_code || ''}`;
   613→        const err = new Error(errMsg);
   614→        err.taskInfo = taskInfo;
   615→        throw err;
   616→      }
   617→      if (status === 4 || status === 'canceled') {
   618→        throw new Error('Render canceled');
   619→      }
   620→
   621→      await new Promise(r => setTimeout(r, intervalMs));
   622→    }
   623→    throw new Error(`Render timeout after ${timeoutMs}ms`);
   624→  }
   625→
   626→  /**
   627→   * Step 8: Download video to local file.
   628→   * @param {string} url - video URL from pollRenderTask
   629→   * @param {string} outPath - local file path
   630→   */
   631→  async downloadVideo(url, outPath) {
   632→    logger.info({ url: url.slice(0, 80), outPath }, 'downloadVideo');
   633→    const res = await axios.get(url, {
   634→      responseType: 'stream',
   635→      timeout: 300000,
   636→      maxContentLength: Infinity,
   637→      maxBodyLength: Infinity,
   638→      headers: { 'User-Agent': DEFAULT_HEADERS['User-Agent'] },
   639→    });
   640→    const ws = fs.createWriteStream(outPath);
   641→    return new Promise((resolve, reject) => {
   642→      res.data.pipe(ws);
   643→      ws.on('finish', () => {
   644→        logger.info({ outPath, size: ws.bytesWritten }, 'downloadVideo: done');
   645→        resolve({ outPath, size: ws.bytesWritten });
   646→      });
   647→      ws.on('error', reject);
   648→      res.data.on('error', reject);
   649→    });
   650→  }

   685→  /**
   686→   * Unwrap CapCut API response.
   687→   * CapCut returns {ret, errmsg, data, log_id} or {code, message, data}.
   688→   */
   689→  _unwrap(resData) {
   690→    if (!resData || typeof resData !== 'object') {
   691→      throw new Error(`Invalid response: ${typeof resData}`);
   692→    }
   693→    // Format 1: {ret: "0", errmsg, data, log_id}
   694→    if ('ret' in resData) {
   695→      const ret = String(resData.ret);
   696→      if (ret !== '0') {
   697→        const err = new Error(`CapCut API error: ret=${ret} errmsg=${resData.errmsg || ''} log_id=${resData.log_id || ''}`);
   698→        err.ret = ret;
   699→        err.errmsg = resData.errmsg;
   700→        err.log_id = resData.log_id;
   701→        throw err;
   702→      }
   703→      return resData.data || {};
   704→    }
   705→    // Format 2: {code: 0, message, data}
   706→    if ('code' in resData) {
   707→      if (resData.code !== 0) {
   708→        const err = new Error(`CapCut API error: code=${resData.code} message=${resData.message || ''}`);
   709→        err.code = resData.code;
   710→        err.message = resData.message;
   711→        throw err;
   712→      }
   713→      return resData.data || {};
   714→    }
   715→    // Already unwrapped
   716→    return resData;
   717→  }
   718→}
