   292→   292→async function findMobileLoginButton() {
   293→   293→  // Strategy 1: find clickable element by text content
   294→   294→  const result = await page.evaluate((texts) => {
   295→   295→    const lowerTexts = texts.map((t) => t.toLowerCase());
   296→   296→    const clickableSelectors = 'button, a, div[role="button"], span[role="button"], [class*="btn" i], [class*="button" i]';
   297→   297→
   298→   298→    const candidates = Array.from(document.querySelectorAll(clickableSelectors));
   299→   299→    for (const el of candidates) {
   300→   300→      // Get text from element AND its children, trimmed
   301→   301→      const text = (el.innerText || el.textContent || '').trim().toLowerCase();
   302→   302→      if (!text || text.length > 100) continue; // skip huge elements / empty
   303→   303→
   304→   304→      const matches = lowerTexts.some((t) => text === t || text.includes(t));
   305→   305→      if (!matches) continue;
   306→   306→
   307→   307→      // Skip elements with very small size (icons, hidden)
   308→   308→      const rect = el.getBoundingClientRect();
   309→   309→      if (rect.width < 100 || rect.height < 20) continue;
   310→   310→      if (rect.top < 0 || rect.left < 0) continue;
   311→   311→
   312→   312→      // Skip if text contains "continue with google" or other non-mobile
   313→   313→      if (text.includes('google') || text.includes('email') || text.includes('tiktok') || text.includes('facebook')) {
   314→   314→        continue;
   315→   315→      }
   316→   316→
   317→   317→      // Found it! Return selector info
   318→   318→      el.setAttribute('data-capcut-mobile-btn', 'true');
   319→   319→      return {
   320→   320→        found: true,
   321→   321→        text: text.slice(0, 80),
   322→   322→        tag: el.tagName.toLowerCase(),
   323→   323→        width: rect.width,
   324→   324→        height: rect.height,
   325→   325→        top: rect.top,
   326→   326→        left: rect.left,
   327→   327→      };
   328→   328→    }
   329→   329→    return { found: false };
   330→   330→  }, MOBILE_BTN_TEXTS);
   331→   331→
   332→   332→  return result;
   333→   333→}

   335→   335→async function findQrCanvasOrImg() {
   336→   336→  // Look for QR code element in modal/popup using multiple strategies
   337→   337→  return await page.evaluate(() => {
   338→   338→    // Strategy 1: canvas with reasonable QR dimensions (200-400px square)
   339→   339→    const canvases = Array.from(document.querySelectorAll('canvas'));
   340→   340→    for (const c of canvases) {
   341→   341→      const rect = c.getBoundingClientRect();
   342→   342→      // QR codes are usually square, 100-400px
   343→   343→      if (Math.abs(rect.width - rect.height) < 20 && rect.width >= 100 && rect.width <= 500) {
   344→   344→        // Skip if it's the CapCut logo or something
   345→   345→        const parent = c.parentElement;
   346→   346→        const parentClass = parent ? parent.className.toLowerCase() : '';
   347→   347→        if (parentClass.includes('logo') || parentClass.includes('icon')) continue;
   348→   348→        return {
   349→   349→          found: true,
   350→   350→          type: 'canvas',
   351→   351→          width: rect.width,
   352→   352→          height: rect.height,
   353→   353→          top: rect.top,
   354→   354→          left: rect.left,
   355→   355→          parentClass: parentClass.slice(0, 80),
   356→   356→        };
   357→   357→      }
   358→   358→    }
   359→   359→
   360→   360→    // Strategy 2: img with qr-ish src or alt
   361→   361→    const imgs = Array.from(document.querySelectorAll('img'));
   362→   362→    for (const i of imgs) {
   363→   363→      const src = (i.src || '').toLowerCase();
   364→   364→      const alt = (i.alt || '').toLowerCase();
   365→   365→      const rect = i.getBoundingClientRect();
   366→   366→      if (rect.width < 80) continue;
   367→   367→      if (src.includes('qr') || alt.includes('qr') ||
   368→   368→          (src.startsWith('data:image/png;base64,') && rect.width >= 100 && Math.abs(rect.width - rect.height) < 30)) {
   369→   369→        return {
   370→   370→          found: true,
   371→   371→          type: 'img',
   372→   372→          width: rect.width,
   373→   373→          height: rect.height,
   374→   374→          top: rect.top,
   375→   375→          left: rect.left,
   376→   376→          src: src.slice(0, 80),
   377→   377→        };
   378→   378→      }
   379→   379→    }
   380→   380→
   381→   381→    // Strategy 3: SVG with class containing qr
   382→   382→    const svgs = Array.from(document.querySelectorAll('svg'));
   383→   383→    for (const s of svgs) {
   384→   384→      const cls = (s.className?.baseVal || s.getAttribute('class') || '').toLowerCase();
   385→   385→      const rect = s.getBoundingClientRect();
   386→   386→      if (cls.includes('qr') && rect.width >= 100) {
   387→   387→        return {
   388→   388→          found: true,
   389→   389→          type: 'svg',
   390→   390→          width: rect.width,
   391→   391→          height: rect.height,
   392→   392→          top: rect.top,
   393→   393→          left: rect.left,
   394→   394→          cls: cls.slice(0, 80),
   395→   395→        };
   396→   396→      }
   397→   397→    }
   398→   398→
   399→   399→    // Strategy 4: any element with qr in class name that's reasonably sized
   400→   400→    const qrEls = Array.from(document.querySelectorAll('[class*="qr" i], [class*="qrcode" i], [class*="scan" i]'));
   401→   401→    for (const el of qrEls) {
   402→   402→      const rect = el.getBoundingClientRect();
   403→   403→      const text = (el.innerText || el.textContent || '').toLowerCase();
   404→   404→      // Skip "Continue with CapCut Mobile" button (it has qr icon class)
   405→   405→      if (text.includes('continue with')) continue;
   406→   406→      if (rect.width < 100 || rect.height < 100) continue;
   407→   407→      return {
   408→   408→        found: true,
   409→   409→        type: 'element',
   410→   410→        width: rect.width,
   411→   411→        height: rect.height,
   412→   412→        top: rect.top,
   413→   413→        left: rect.left,
   414→   414→        cls: (el.className || '').toString().slice(0, 80),
   415→   415→      };
   416→   416→    }
   417→   417→
   418→   418→    return { found: false };
   419→   419→  });
   420→   420→}
   421→   421→
   422→   422→let mobileBtnClicked = false;
   423→   423→let mobileBtnInfo = null;
   424→   424→

   441→   441→        await new Promise((r) => setTimeout(r, 200));
   442→   442→        // Real click via Puppeteer mouse (sends trusted OS-level event)
   443→   443→        await page.mouse.click(centerX, centerY, { delay: 50 });
   444→   444→        mobileBtnClicked = true;
   445→   445→        logger.info({ x: centerX, y: centerY }, 'Real mouse click sent to button center');
   446→   446→      } catch (e) {
   447→   447→        logger.warn({ err: e.message }, 'Mouse click failed, trying element.click() fallback');
   448→   448→        // Fallback: use elementHandle.click() (Puppeteer's native click)
   449→   449→        try {
   450→   450→          const handle = await page.$('[data-capcut-mobile-btn="true"]');
   451→   451→          if (handle) {
   452→   452→            await handle.click();
   453→   453→            mobileBtnClicked = true;
   454→   454→            logger.info('ElementHandle.click() fallback succeeded');
   455→   455→          }
   456→   456→        } catch (e2) {
   457→   457→          logger.warn({ err: e2.message }, 'ElementHandle.click() also failed');
   458→   458→        }
   459→   459→      }
   460→   460→
   461→   461→      // Remove the data attribute for cleanliness
   462→   462→      await page.evaluate(() => {
   463→   463→        const btn = document.querySelector('[data-capcut-mobile-btn="true"]');
   464→   464→        if (btn) btn.removeAttribute('data-capcut-mobile-btn');
   465→   465→      });
   466→   466→
   467→   467→      if (mobileBtnClicked) break;
   468→   468→    }
   469→   469→  } catch (e) {
   470→   470→    logger.warn({ err: e.message, attempt }, 'Error finding mobile button');
   471→   471→  }
   472→   472→  if (!mobileBtnClicked) {
   473→   473→    if (attempt === 0) logger.warn('CapCut Mobile button not found yet, retrying...');
   474→   474→    await new Promise((r) => setTimeout(r, 2000));
   475→   475→  }
   476→   476→}
   477→   477→
   478→   478→if (!mobileBtnClicked) {
   479→   479→  logger.warn('Could not find "Continue with CapCut Mobile" button after 8 attempts.');
   480→   480→  logger.info('Continuing anyway — will try to detect QR if it auto-appears...');
   481→   481→} else {
   482→   482→  logger.info('Waiting for QR code modal to render...');
   483→   483→  await new Promise((r) => setTimeout(r, 3000));
   484→   484→
   485→   485→  // Diagnostic: check what changed after click
   486→   486→  const diag = await page.evaluate(() => {
   487→   487→    const allPages = window.length; // number of frames
   488→   488→    const canvases = document.querySelectorAll('canvas').length;
   489→   489→    const iframes = document.querySelectorAll('iframe').length;
   490→   490→    const dialogs = document.querySelectorAll('[role="dialog"], [class*="modal" i], [class*="popup" i], [class*="dialog" i]').length;
   491→   491→    // Check for any newly-visible overlay

   493→   493→      .filter((d) => {

   496→   496→        return (
   497→   497→          (s.position === 'fixed' || s.position === 'absolute') &&
   498→   498→          rect.width > 200 &&
   499→   499→          rect.height > 200 &&
   500→   500→          s.zIndex !== 'auto' &&
   501→   501→          parseInt(s.zIndex || '0', 10) > 100
   502→   502→        );
   503→   503→      }).length;
   504→   504→    return { allPages, canvases, iframes, dialogs, overlays, url: location.href };
   505→   505→  }).catch(() => ({}));
   506→   506→  logger.info({ diag }, 'Post-click diagnostics (canvases, iframes, dialogs, overlays)');
   507→   507→

   509→   509→  const iframeInfos = await page.evaluate(() => {
   510→   510→    return Array.from(document.querySelectorAll('iframe')).map((f) => {
   511→   511→      const r = f.getBoundingClientRect();

   564→   564→
   565→   565→// ====== Polling: detect login + capture QR ======
   566→   566→const startTime = Date.now();
   567→   567→let lastScreenshotTime = 0;
   568→   568→let qrFound = false;
   569→   569→let tries = 0;
   570→   570→
   571→   571→async function checkLogin() {
   572→   572→  tries++;
   573→   573→  const elapsedMs = Date.now() - startTime;
   574→   574→
   575→   575→  if (elapsedMs > MAX_WAIT_MS) {
   576→   576→    logger.warn({ elapsedMs }, `Timeout ${MAX_WAIT_MS / 1000}s. No login detected.`);
   577→   577→    return 'timeout';
   578→   578→  }
   579→   579→
   580→   580→  try {
   581→   581→    // Use popupPage if it exists (where QR code lives), otherwise main page
   582→   582→    const activePage = popupPage && !popupPage.isClosed() ? popupPage : page;
   583→   583→    const currentUrl = activePage.url();
   584→   584→
   585→   585→    // Cek 1: URL berubah dari /login atau /scan-qr-code (redirect ke home/dashboard)
   586→   586→    const stillOnLogin = /\/login|\/scan-qr-code/i.test(currentUrl);
   587→   587→
   588→   588→    // Cek 2: cookies session (check both pages, cookies are shared per profile)
   589→   589→    const cookies = await page.cookies();
   590→   590→    const sessionCookie = cookies.find(
   591→   591→      (c) =>
   592→   592→        /session|token|uid|passport|login|sid|tt_csrf|s_v_web_id/i.test(c.name) && c.value && c.value.length > 10
   593→   593→    );
   594→   594→
   595→   595→    // Cek 3: avatar / user element di header (check main page)
   596→   596→    let avatarFound = false;
   597→   597→    if (!stillOnLogin) {
   598→   598→      try {
   599→   599→        avatarFound = await page.$(
   600→   600→          '[class*="avatar" i], [data-e2e*="user" i], [data-testid*="user" i], [class*="user-info" i], [class*="header-user" i], [class*="account" i]'
   601→   601→        ) ? true : false;
   602→   602→      } catch (_) {}
   603→   603→    }
   604→   604→
   605→   605→    if (sessionCookie && (!stillOnLogin || avatarFound)) {
   606→   606→      logger.info({ url: currentUrl, cookie: sessionCookie?.name, avatarFound }, 'LOGIN SUCCESS detected!');
   607→   607→      return 'success';
   608→   608→    }
   609→   609→
   610→   610→    // Screenshot QR code — prefer popupPage (where QR actually is)
   611→   611→    const now = Date.now();
   612→   612→    if (now - lastScreenshotTime > 1500) {
   613→   613→      try {
   614→   614→        if (popupPage && !popupPage.isClosed()) {
   615→   615→          // Screenshot the popup page (QR code lives here!)
   616→   616→          await popupPage.screenshot({ path: qrLatestPath, fullPage: false }).catch(() => {});
   617→   617→          if (!qrFound) {
   618→   618→            logger.info({ popupUrl: popupPage.url() }, 'Capturing QR from popup page');
   619→   619→            qrFound = true;
   620→   620→          }
   621→   621→        } else {
   622→   622→          // Fallback: try to find QR element on main page
   623→   623→          const found = await findQrCanvasOrImg();
   624→   624→          if (found && found.found) {
   625→   625→            if (!qrFound) {
   626→   626→              logger.info(
   627→   627→                { type: found.type, w: Math.round(found.width), h: Math.round(found.height) },
   628→   628→                'QR code element detected on main page, capturing...'
   629→   629→              );
   630→   630→              qrFound = true;
   631→   631→            }
   632→   632→            await page.screenshot({
   633→   633→              path: qrLatestPath,
   634→   634→              clip: {
   635→   635→                x: Math.max(0, found.left - 30),
   636→   636→                y: Math.max(0, found.top - 30),
   637→   637→                width: found.width + 60,
   638→   638→                height: found.height + 60,
   639→   639→              },
   640→   640→            });
   641→   641→          } else {
   642→   642→            await page.screenshot({ path: qrLatestPath, fullPage: false });
   643→   643→          }
   644→   644→        }
   645→   645→        lastScreenshotTime = now;
   646→   646→
   647→   647→        // Save periodic debug screenshot
   648→   648→        if (tries % 5 === 0) {
   649→   649→          const debugPath = path.join(screenshotDir, `debug-${tries}.png`);
   650→   650→          await activePage.screenshot({ path: debugPath, fullPage: false }).catch(() => {});
   651→   651→        }
   652→   652→      } catch (e) {
   653→   653→        logger.warn({ err: e.message }, 'Screenshot error');
   654→   654→      }
   655→   655→    }
   656→   656→
   657→   657→    if (tries % 6 === 0) {
   658→   658→      logger.info(
   659→   659→        { tries, elapsedS: Math.round(elapsedMs / 1000), url: currentUrl, qrFound, hasSessionCookie: !!sessionCookie, mobileBtnClicked, popupAlive: popupPage && !popupPage.isClosed() },
   660→   660→        'Waiting for login...'
   661→   661→      );
   662→   662→    }
   663→   663→  } catch (e) {
   664→   664→    logger.warn({ err: e.message }, 'Polling error');
   665→   665→  }
   666→   666→
   667→   667→  return 'continue';
   668→   668→}
   669→   669→
   670→   670→const pollLoop = async () => {
   671→   671→  while (true) {
   672→   672→    const result = await checkLogin();
   673→   673→    if (result === 'success') {
   674→   674→      // Beri delay sedikit untuk pastikan session tersimpan
   675→   675→      await new Promise((r) => setTimeout(r, 2000));
   676→   676→      try {
   677→   677→        await browser.close();
   678→   678→      } catch (_) {}
   679→   679→      httpServer.close();
   680→   680→      logger.info(`✓ Session saved to ${userDataDir}`);
   681→   681→      logger.info('Set CAPCUT_USER_DATA_DIR=./.capcut-profile di .env untuk reuse session ini.');
   682→   682→      process.exit(0);
   683→   683→    }
   684→   684→    if (result === 'timeout') {
   685→   685→      try {
   686→   686→        await browser.close();
   687→   687→      } catch (_) {}
   688→   688→      httpServer.close();
   689→   689→      process.exit(1);
   690→   690→    }
   691→   691→    await new Promise((r) => setTimeout(r, POLL_INTERVAL_MS));
   692→   692→  }
   693→   693→};
   694→   694→
   695→   695→pollLoop().catch((e) => {
   696→   696→  logger.error({ err: e.message }, 'Fatal error in poll loop');
   697→   697→  try { browser.close(); } catch (_) {}
   698→   698→  httpServer.close();
   699→   699→  process.exit(1);
   700→   700→});
   701→   701→
   702→   702→// Graceful shutdown
   703→   703→const shutdown = async (sig) => {
   704→   704→  logger.info({ sig }, 'Received signal, closing browser...');
   705→   705→  try { await browser.close(); } catch (_) {}
   706→   706→  httpServer.close();
   707→   707→  process.exit(0);
   708→   708→};
   709→   709→process.on('SIGINT', () => shutdown('SIGINT'));
   710→   710→process.on('SIGTERM', () => shutdown('SIGTERM'));
